function submitPayment(payType) { if (busy) return; if (!idToken) { setStatus('ไม่พบ ID Token กรุณาเปิดผ่าน LIFF ใหม่', 'error'); return; } const month = $('paymentMonth').value; if (!/^\d{4}-(0[1-9]|1[0-2])$/.test(month)) { setStatus('กรุณาเลือกเดือนที่ต้องการบันทึก', 'error'); return; } busy = true; setButtons(false); setStatus('กำลังบันทึกและตรวจสอบข้อมูลบนเซิร์ฟเวอร์...', 'info'); let finished = false; const watchdog = setTimeout(function() { if (finished) return; finished = true; busy = false; // ไม่เปิดปุ่มทันที เพราะเซิร์ฟเวอร์อาจบันทึกสำเร็จแล้ว setStatus( 'เซิร์ฟเวอร์ตอบกลับช้าเกิน 30 วินาที ' + 'กรุณาตรวจสอบ Google Sheets และ Apps Script > Executions ' + 'ก่อนลองบันทึกอีกครั้ง', 'error' ); }, 30000); google.script.run .withSuccessHandler(function(result) { if (finished) return; finished = true; clearTimeout(watchdog); busy = false; if (!result) { setStatus( 'เซิร์ฟเวอร์ไม่ส่งผลยืนยันกลับมา กรุณาตรวจสอบชีต', 'error' ); return; } if (result.duplicate) { setButtons(true); setStatus( '⚠️ ' + (result.message || 'มีรายการนี้แล้ว') + '\nเดือน: ' + (result.month || month) + '\nวิธีจ่ายเดิม: ' + (result.payType || '-'), 'error' ); return; } if (result.success !== true) { setButtons(true); setStatus( result.message || 'เซิร์ฟเวอร์ไม่ยืนยันการบันทึก', 'error' ); return; } setStatus( '✅ บันทึกสำเร็จ' + '\nชื่อ: ' + (result.name || '-') + '\nเดือน: ' + (result.month || month) + '\nวิธีจ่าย: ' + (result.payType || payType) + '\nเวลาบันทึก: ' + (result.recordedAt || '-'), 'success' ); if (liff.isInClient()) { setTimeout(function() { liff.closeWindow(); }, 1500); } else { setButtons(true); } }) .withFailureHandler(function(error) { if (finished) return; finished = true; clearTimeout(watchdog); busy = false; setButtons(true); console.error('savePayment error:', error); setStatus( '❌ บันทึกไม่สำเร็จ: ' + (error && error.message ? error.message : 'ไม่ทราบสาเหตุ') + '\nกรุณาตรวจสอบชีตก่อนลองใหม่', 'error' ); }) .savePayment({ idToken: idToken, month: month, payType: payType }); }
